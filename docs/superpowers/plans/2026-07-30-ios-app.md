# Laces Out iOS App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A SwiftUI + WKWebView iOS app (`apps/ios`) that wraps the Laces Out web UI with a native tab bar, onboarding, Face ID lock, share sheet, and real APNs push — plus the contained API/worker/web changes that support it.

**Architecture:** The server work adds an `apns` transport to the existing push pipeline (contracts union → `push_subscriptions.transport` column → a new `@fantasy/apns` HTTP/2 client → transport-routing senders in both the API's `PushSubscriptionService` and the worker's `NotificationSender`). The web work adds a `native-bridge` lib the shell talks to. The iOS work is an XcodeGen-defined SwiftUI shell with one `WKWebView` per tab sharing one cookie store, authored on Linux and compiled at explicit Mac checkpoints.

**Tech Stack:** TypeScript (Fastify API, worker, Next.js web, vitest, drizzle), Swift 5.10 / SwiftUI / WebKit / XCTest, XcodeGen, APNs token-based auth via `node:crypto` + `node:http2` (no new npm dependencies).

**Spec:** `docs/superpowers/specs/2026-07-30-ios-app-design.md`

`packages/apns/src/index.ts`:

```ts
import { createPrivateKey, sign } from "node:crypto";

/**
 * APNs delivery over token-based auth (`.p8` key), shared by the API's test sender and the
 * worker's notification sweep. This module is the only code that touches a device token value:
 * tokens arrive as opaque strings and leave only on their way to Apple, never to a log line.
 */

export interface ApnsIdentity {
  readonly teamId: string;
  readonly keyId: string;
  readonly privateKeyPem: string;
}

export type ApnsEnvironment = "development" | "production";

export type ApnsSendOutcome = "sent" | "expired" | "failed";

export interface ApnsNotification {
  readonly title: string;
  readonly body: string;
  /** Application-relative deep link the app opens on tap. */
  readonly url: string;
  /** Collapses a superseded notification on the device instead of stacking a second one. */
  readonly tag: string;
}

export interface ApnsHttpRequest {
  readonly authority: string;
  readonly path: string;
  readonly headers: Record<string, string>;
  readonly body: string;
}

export interface ApnsHttpResponse {
  readonly status: number;
  readonly reason: string | null;
}

export type ApnsPerform = (request: ApnsHttpRequest) => Promise<ApnsHttpResponse>;

/**
 * Mints the ES256 provider JWT by hand: `dsaEncoding: "ieee-p1363"` makes node produce the raw
 * r||s signature JWS requires, which keeps this dependency-free.
 */
export function mintApnsToken(identity: ApnsIdentity, nowSeconds: number): string {
  const header = Buffer.from(JSON.stringify({ alg: "ES256", kid: identity.keyId })).toString(
    "base64url",
  );
  const claims = Buffer.from(JSON.stringify({ iss: identity.teamId, iat: nowSeconds })).toString(
    "base64url",
  );
  const signingInput = `${header}.${claims}`;
  const key = createPrivateKey(identity.privateKeyPem);
  const signature = sign("sha256", Buffer.from(signingInput), { key, dsaEncoding: "ieee-p1363" });
  return `${signingInput}.${signature.toString("base64url")}`;
}

const APNS_HOSTS: Record<ApnsEnvironment, string> = {
  development: "api.sandbox.push.apple.com",
  production: "api.push.apple.com",
};

/** Apple accepts provider tokens for 20–60 minutes; re-mint comfortably inside that window. */
const TOKEN_LIFETIME_SECONDS = 50 * 60;

/** Reasons that mean the token will never work again, so the row must be pruned, not retried. */
const EXPIRED_REASONS: ReadonlySet<string> = new Set([
  "BadDeviceToken",
  "Unregistered",
  "DeviceTokenNotForTopic",
  "ExpiredToken",
]);

export interface ApnsClientOptions {
  readonly identity: ApnsIdentity;
  readonly bundleId: string;
  readonly now?: () => Date;
  readonly perform?: ApnsPerform;
}

export interface ApnsClient {
  send(
    deviceToken: string,
    environment: ApnsEnvironment,
    notification: ApnsNotification,
  ): Promise<ApnsSendOutcome>;
}

export function createApnsClient(options: ApnsClientOptions): ApnsClient {
  const now = options.now ?? (() => new Date());
  const perform = options.perform ?? performHttp2Request;
  let minted: { readonly token: string; readonly atSeconds: number } | undefined;

  function bearerToken(): string {
    const nowSeconds = Math.floor(now().getTime() / 1000);
    if (!minted || nowSeconds - minted.atSeconds >= TOKEN_LIFETIME_SECONDS) {
      minted = { token: mintApnsToken(options.identity, nowSeconds), atSeconds: nowSeconds };
    }
    return minted.token;
  }

  return {
    async send(deviceToken, environment, notification): Promise<ApnsSendOutcome> {
      try {
        const response = await perform({
          authority: `https://${APNS_HOSTS[environment]}`,
          path: `/3/device/${deviceToken}`,
          headers: {
            authorization: `bearer ${bearerToken()}`,
            "apns-topic": options.bundleId,
            "apns-push-type": "alert",
            "apns-priority": "10",
            "apns-collapse-id": notification.tag.slice(0, 64),
          },
          body: JSON.stringify({
            aps: {
              alert: { title: notification.title, body: notification.body },
              sound: "default",
              "thread-id": notification.tag,
            },
            url: notification.url,
            tag: notification.tag,
          }),
        });
        if (response.status === 200) return "sent";
        if (response.status === 410) return "expired";
        if (response.status === 400 && response.reason !== null && EXPIRED_REASONS.has(response.reason)) {
          return "expired";
        }
        return "failed";
      } catch {
        return "failed";
      }
    },
  };
}

/** One short-lived HTTP/2 session per send; league-scale volume never needs connection reuse. */
async function performHttp2Request(request: ApnsHttpRequest): Promise<ApnsHttpResponse> {
  const { connect } = await import("node:http2");
  return new Promise((resolve, reject) => {
    const session = connect(request.authority);
    session.on("error", reject);
    const stream = session.request({
      ":method": "POST",
      ":path": request.path,
      "content-type": "application/json",
      ...request.headers,
    });
    let status = 0;
    const chunks: Buffer[] = [];
    stream.on("response", (headers) => {
      status = Number(headers[":status"] ?? 0);
    });
    stream.on("data", (chunk: Buffer) => chunks.push(chunk));
    stream.on("error", reject);
    stream.on("end", () => {
      session.close();
      let reason: string | null = null;
      try {
        const parsed: unknown = JSON.parse(Buffer.concat(chunks).toString("utf8"));
        if (
          parsed !== null &&
          typeof parsed === "object" &&
          "reason" in parsed &&
          typeof parsed.reason === "string"
        ) {
          reason = parsed.reason;
        }
      } catch {
        // A 200 has an empty body; no reason to extract.
      }
      resolve({ status, reason });
    });
    stream.end(request.body);
  });
}
```

```ts
/**
 * Detection and messaging for the native iOS shell. The shell injects `window.__lacesOutNative`
 * at document start and registers a `lacesout` WebKit message handler. Everything here no-ops
 * cleanly in an ordinary browser (and during SSR), so callers may invoke it unconditionally and
 * fall back on their web behavior when it answers false.
 */

export interface NativeShellInfo {
  readonly version: string;
  readonly capabilities: readonly string[];
}

export interface SharePayload {
  readonly title: string;
  readonly text: string;
  readonly url?: string;
}

interface LacesOutMessageHandler {
  postMessage(message: unknown): void;
}

declare global {
  interface Window {
    __lacesOutNative?: NativeShellInfo;
    webkit?: { messageHandlers?: { lacesout?: LacesOutMessageHandler } };
  }
}

function shellWithCapability(
  capability: string,
): { readonly handler: LacesOutMessageHandler } | null {
  if (typeof window === "undefined") return null;
  const shell = window.__lacesOutNative;
  const handler = window.webkit?.messageHandlers?.lacesout;
  if (!shell || !handler || !shell.capabilities.includes(capability)) return null;
  return { handler };
}

export function isNativeShell(): boolean {
  if (typeof window === "undefined") return false;
  return Boolean(window.__lacesOutNative && window.webkit?.messageHandlers?.lacesout);
}

/** Opens the iOS share sheet. Returns false when not embedded so callers fall back to copy. */
export function nativeShare(payload: SharePayload): boolean {
  const shell = shellWithCapability("share");
  if (!shell) return false;
  shell.handler.postMessage({ type: "share", ...payload });
  return true;
}

/** Opens the native app-settings sheet (server URL, Face ID lock). */
export function openNativeSettings(): boolean {
  const shell = shellWithCapability("settings");
  if (!shell) return false;
  shell.handler.postMessage({ type: "openSettings" });
  return true;
}
```

```swift
import Foundation

/// The one persisted pointer at a Laces Out instance. Everything else — session, league state,
/// preferences — lives on the server or in the webview's cookie store.
struct ServerConfig: Equatable {
    let baseURL: URL

    static let defaultsKey = "serverBaseURL"

    /// Normalizes member input: assumes https when no scheme is typed, keeps an explicit port,
    /// and drops any path, query, or fragment so the stored value is origin-only.
    static func normalize(_ input: String) -> URL? {
        var text = input.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !text.isEmpty else { return nil }
        if !text.contains("://") { text = "https://" + text }
        guard var components = URLComponents(string: text),
              let scheme = components.scheme?.lowercased(), scheme == "https" || scheme == "http",
              let host = components.host, !host.isEmpty
        else { return nil }
        components.scheme = scheme
        components.path = ""
        components.query = nil
        components.fragment = nil
        return components.url
    }

    static func load(from defaults: UserDefaults = .standard) -> ServerConfig? {
        guard let raw = defaults.string(forKey: defaultsKey), let url = URL(string: raw) else { return nil }
        return ServerConfig(baseURL: url)
    }

    func save(to defaults: UserDefaults = .standard) {
        defaults.set(baseURL.absoluteString, forKey: Self.defaultsKey)
    }

    func url(path: String) -> URL {
        var components = URLComponents(url: baseURL, resolvingAgainstBaseURL: false)!
        components.path = path
        return components.url!
    }
}

@MainActor
final class ServerStore: ObservableObject {
    @Published private(set) var config: ServerConfig?

    init() {
        config = ServerConfig.load()
    }

    func apply(_ newConfig: ServerConfig) {
        newConfig.save()
        config = newConfig
    }

    func reset() {
        UserDefaults.standard.removeObject(forKey: ServerConfig.defaultsKey)
        config = nil
    }
}
```

```swift
import Foundation

/// Same-origin pages and the Yahoo OAuth hop stay inside the webview; everything else leaves the
/// app. Pure so the decision table is unit-testable.
struct NavigationPolicy {
    let serverHost: String

    enum Decision: Equatable {
        case allow
        case openExternally
    }

    func decide(url: URL) -> Decision {
        guard let scheme = url.scheme?.lowercased(), scheme == "https" || scheme == "http" else {
            return .openExternally
        }
        guard let host = url.host?.lowercased() else { return .allow }
        if host == serverHost.lowercased() { return .allow }
        if host == "yahoo.com" || host.hasSuffix(".yahoo.com") { return .allow }
        return .openExternally
    }
}
```

```swift
import WebKit

/// One WKWebView per tab so scroll and in-page state survive tab switches; one shared default
/// website data store so all tabs (and Task 13's registration call) see the same session cookie.
@MainActor
final class WebViewStore: NSObject, ObservableObject {
    let config: ServerConfig
    let policy: NavigationPolicy
    var onBridgeMessage: ((BridgeMessage) -> Void)?
    var onSessionCookie: (() -> Void)?

    private var webViews: [AppTab: WKWebView] = [:]

    static let sessionCookieName = "fantasy_session"
    static let bootstrapCapabilities = #"["share", "settings"]"#

    init(config: ServerConfig) {
        self.config = config
        self.policy = NavigationPolicy(serverHost: config.baseURL.host ?? "")
    }

    func webView(for tab: AppTab) -> WKWebView {
        if let existing = webViews[tab] { return existing }
        let configuration = WKWebViewConfiguration()
        configuration.websiteDataStore = .default()
        let version = Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "0"
        configuration.applicationNameForUserAgent = "LacesOutiOS/\(version)"
        let bootstrap = """
        window.__lacesOutNative = { version: "\(version)", capabilities: \(Self.bootstrapCapabilities) };
        """
        configuration.userContentController.addUserScript(
            WKUserScript(source: bootstrap, injectionTime: .atDocumentStart, forMainFrameOnly: true)
        )
        configuration.userContentController.add(BridgeRelay(store: self), name: "lacesout")
        let webView = WKWebView(frame: .zero, configuration: configuration)
        webView.allowsBackForwardNavigationGestures = true
        webView.isOpaque = false
        webView.load(URLRequest(url: config.url(path: tab.path)))
        webViews[tab] = webView
        return webView
    }

    func load(path: String, in tab: AppTab) {
        webView(for: tab).load(URLRequest(url: config.url(path: path)))
    }

    func checkForSessionCookie() {
        WKWebsiteDataStore.default().httpCookieStore.getAllCookies { [weak self] cookies in
            guard cookies.contains(where: { $0.name == Self.sessionCookieName }) else { return }
            self?.onSessionCookie?()
        }
    }
}

/// Message-handler indirection that avoids a retain cycle between the user content controller
/// and the store. Parsing lives in `BridgeMessage.parse` (Task 11) so it stays testable.
private final class BridgeRelay: NSObject, WKScriptMessageHandler {
    weak var store: WebViewStore?

    init(store: WebViewStore) {
        self.store = store
    }

    func userContentController(_ controller: WKUserContentController, didReceive message: WKScriptMessage) {
        guard let parsed = BridgeMessage.parse(message.body) else { return }
        Task { @MainActor in
            self.store?.onBridgeMessage?(parsed)
        }
    }
}
```

```swift
import SwiftUI
import WebKit

/// Hosts one tab's webview: navigation policy, pull-to-refresh, load-failure reporting, and
/// session-cookie detection after each successful page load.
struct WebViewContainer: UIViewRepresentable {
    let store: WebViewStore
    let tab: AppTab
    @Binding var loadFailed: Bool

    func makeCoordinator() -> Coordinator {
        Coordinator(container: self)
    }

    func makeUIView(context: Context) -> WKWebView {
        let webView = store.webView(for: tab)
        webView.navigationDelegate = context.coordinator
        webView.uiDelegate = context.coordinator
        let refresh = UIRefreshControl()
        refresh.addTarget(context.coordinator, action: #selector(Coordinator.refresh(_:)), for: .valueChanged)
        webView.scrollView.refreshControl = refresh
        return webView
    }

    func updateUIView(_ webView: WKWebView, context: Context) {
        context.coordinator.container = self
    }

    final class Coordinator: NSObject, WKNavigationDelegate, WKUIDelegate {
        var container: WebViewContainer

        init(container: WebViewContainer) {
            self.container = container
        }

        @objc func refresh(_ sender: UIRefreshControl) {
            container.store.webView(for: container.tab).reload()
        }

        func webView(
            _ webView: WKWebView,
            decidePolicyFor navigationAction: WKNavigationAction,
            decisionHandler: @escaping (WKNavigationActionPolicy) -> Void
        ) {
            guard let url = navigationAction.request.url else {
                decisionHandler(.allow)
                return
            }
            switch container.store.policy.decide(url: url) {
            case .allow:
                decisionHandler(.allow)
            case .openExternally:
                decisionHandler(.cancel)
                UIApplication.shared.open(url)
            }
        }

        func webView(_ webView: WKWebView, didFinish navigation: WKNavigation!) {
            webView.scrollView.refreshControl?.endRefreshing()
            container.loadFailed = false
            container.store.checkForSessionCookie()
        }

        func webView(_ webView: WKWebView, didFailProvisionalNavigation navigation: WKNavigation!, withError error: Error) {
            webView.scrollView.refreshControl?.endRefreshing()
            container.loadFailed = true
        }

        /// target=_blank: same-origin popups load in place, anything else leaves the app.
        func webView(
            _ webView: WKWebView,
            createWebViewWith configuration: WKWebViewConfiguration,
            for navigationAction: WKNavigationAction,
            windowFeatures: WKWindowFeatures
        ) -> WKWebView? {
            guard let url = navigationAction.request.url else { return nil }
            switch container.store.policy.decide(url: url) {
            case .allow:
                webView.load(URLRequest(url: url))
            case .openExternally:
                UIApplication.shared.open(url)
            }
            return nil
        }
    }
}
```

`apps/ios/LacesOut/BridgeMessage.swift`:

```swift
import Foundation

/// The typed contract behind `window.webkit.messageHandlers.lacesout`. Anything unparseable is
/// dropped silently: the page may be newer or older than the shell and neither should break.
enum BridgeMessage: Equatable {
    case share(title: String, text: String, url: String?)
    case openSettings

    static func parse(_ body: Any) -> BridgeMessage? {
        guard let dictionary = body as? [String: Any], let type = dictionary["type"] as? String else {
            return nil
        }
        switch type {
        case "share":
            guard let title = dictionary["title"] as? String, let text = dictionary["text"] as? String else {
                return nil
            }
            return .share(title: title, text: text, url: dictionary["url"] as? String)
        case "openSettings":
            return .openSettings
        default:
            return nil
        }
    }
}
```

```swift
import LocalAuthentication
import SwiftUI

/// Optional app lock: content is covered the moment the app leaves the foreground and stays
/// covered until Face ID / Touch ID / passcode succeeds. The web session is untouched either way.
@MainActor
final class AppLock: ObservableObject {
    @AppStorage("appLockEnabled") private var enabled = false
    @Published var locked = false

    func handleScenePhase(_ phase: ScenePhase) {
        if enabled && phase == .background {
            locked = true
        }
    }

    func unlock() async {
        let context = LAContext()
        var error: NSError?
        guard context.canEvaluatePolicy(.deviceOwnerAuthentication, error: &error) else {
            // A device with no passcode can't authenticate; leaving it locked would brick the app.
            locked = false
            return
        }
        let success = (try? await context.evaluatePolicy(
            .deviceOwnerAuthentication,
            localizedReason: "Unlock Laces Out"
        )) ?? false
        if success {
            locked = false
        }
    }
}
```

```swift
import Foundation

struct DeepLink: Equatable {
    let tab: AppTab
    let path: String
}

/// Maps a notification's `url` payload key to a tab and page. Only server-relative paths are
/// honored: an absolute URL in a payload never steers the webview.
enum DeepLinkRouter {
    static func deepLink(fromNotificationUserInfo userInfo: [AnyHashable: Any]) -> DeepLink? {
        guard let path = userInfo["url"] as? String, path.hasPrefix("/"), !path.hasPrefix("//") else {
            return nil
        }
        return DeepLink(tab: .owning(path: path), path: path)
    }
}
```

`apps/ios/LacesOut/PushRegistrar.swift`:

```swift
import UIKit
import UserNotifications
import WebKit

/// Asks for notification permission only after a signed-in session exists, then registers the
/// APNs token with the member's own instance. Every failure here is silent by design: alerts are
/// a bonus, never a blocker.
@MainActor
final class PushRegistrar: NSObject, ObservableObject {
    var config: ServerConfig?
    private var sessionHandled = false

    func handleSessionDetected() {
        guard !sessionHandled else { return }
        sessionHandled = true
        Task {
            let center = UNUserNotificationCenter.current()
            let settings = await center.notificationSettings()
            switch settings.authorizationStatus {
            case .notDetermined:
                let granted = (try? await center.requestAuthorization(options: [.alert, .badge, .sound])) ?? false
                guard granted else { return }
            case .denied:
                return
            default:
                break
            }
            UIApplication.shared.registerForRemoteNotifications()
        }
    }

    func handleDeviceToken(_ token: Data) {
        guard let config else { return }
        let hex = token.map { String(format: "%02x", $0) }.joined()
        Task { await Self.register(token: hex, config: config) }
    }

    static func register(token: String, config: ServerConfig) async {
        // The session lives in the webview's cookie store; copy it onto this native request. The
        // API checks Origin on mutations, so name the instance explicitly.
        let cookies = await WKWebsiteDataStore.default().httpCookieStore.allCookies()
        guard let session = cookies.first(where: { $0.name == WebViewStore.sessionCookieName }) else {
            return
        }
        var request = URLRequest(url: config.url(path: "/v1/push/subscriptions"))
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue("\(session.name)=\(session.value)", forHTTPHeaderField: "Cookie")
        request.setValue(config.baseURL.absoluteString, forHTTPHeaderField: "Origin")
        #if DEBUG
        let environment = "development"
        #else
        let environment = "production"
        #endif
        let body: [String: Any] = [
            "transport": "apns",
            "deviceToken": token,
            "environment": environment,
            "label": "\(UIDevice.current.model) (app)",
        ]
        request.httpBody = try? JSONSerialization.data(withJSONObject: body)
        _ = try? await URLSession.shared.data(for: request)
    }
}
```
